\section{Proof of the radial law}\label{sec:layer3}\label{sec:unit-circle}\label{sec:moving-radii}
Sections~\ref{sec:radial}--\ref{sec:local-roots} control the roots of
$f_N$ along the sparse subsequence of degrees, and root matching carries
most of them to every degree of the block. The count at $1+x/n$ then
differs from the count at $1+x/N$ only through roots near the target
circle, unselected roots and the degree difference, and each of these is
small. We prove Theorem~\ref{thm:radial-profile} for one $x$ at a time,
and then for all $x$ by monotonicity; Corollary~\ref{thm:stronglaw} is
the case $x=0$, since $\Phi(0)=1/2$.

\begin{proof}[Proof of Theorem~\ref{thm:radial-profile}]
The idea is to compare every degree $n$ in a block
$j^8\le n\le(j+1)^8$ with the first degree $N=j^8$ of the block, by root
matching (Corollary~\ref{cor:T4-two-radii}) at the radii $1+x/N$ and
$1+x/n$. A root of $f_N$ is matched unless it lies outside the annulus
$||z|-1|\le K/N$ or has a small derivative there, and a matched root can
cross between the two target circles only if its disk meets the annulus
between them.

\emph{Step 1: parameters.}

degrees $N=j^8\le n\le(j+1)^8=N+m_j$. Table~\ref{tab:unit-conditions}
lists the parameters, the conditions they must satisfy and where they
are used. The annulus is $||z|-1|\le K/N$, and $L=N^{1/64}$ as in
Section~\ref{sec:derivative-detection}.

The conditions hold for all large $j$ because the block exponent $8$
makes $m_j$ of order $N^{7/8}$, and hence $a$ of order
$N^{7/16}\sqrt{\log N}$, while $d_0=N^{3/2-1/64}$.
With blocks $j^\beta$, $m_j$ would be of order $N^{1-1/\beta}$, and
$2a/d_0$ and $4M_2a/d_0^2$ of orders $N^{(1-1/\beta)/2-95/64}$ and
$N^{(1-1/\beta)/2-15/32}$, up to logarithmic factors, so Steps~2 and~4
would need $(1-1/\beta)/2<15/32$, that is, $\beta<16$. The slowest
exceptional probability in Step~6, of order $N^{-5/16}(\log N)^2$, is
summable along $N=j^\beta$ only if $\beta>16/5$.
Proposition~\ref{prop:sparse-exponents} treats other block exponents.
For $j\ge60$ we have
\begin{align}
 m_j&=(j+1)^8-j^8\le8(j+1)^7=8\Bigl(1+\frac1j\Bigr)^7N^{7/8}\nonumber\\
 &\le8\Bigl(\frac{61}{60}\Bigr)^7N^{7/8}\le9N^{7/8},
 \label{eq:unit-block-length}\\
 a&\le45e^{2(K+2)}N^{7/16}\sqrt{\log N},\label{eq:unit-amplitude}\\
 \frac{2a}{d_0}
 &=\frac{2a}{N^{3/2-1/64}}
 \le90e^{2(K+2)}N^{7/16+1/64-3/2}\sqrt{\log N}\nonumber\\
 &=90e^{2(K+2)}N^{-67/64}\sqrt{\log N},\label{eq:unit-radius}\\
 \frac{4Na}{d_0}&\le180e^{2(K+2)}N^{-3/64}\sqrt{\log N}\longrightarrow0,
 \label{eq:unit-radius-scaled}\\
 \frac{4M_2a}{d_0^2}
 &=\frac{400e^{K+4}N^{5/2}\sqrt{\log N}\,a}{N^{3-1/32}}
 \le180\cdot100e^{K+4}\,e^{2(K+2)}N^{7/16+1/32-1/2}\log N\nonumber\\
 &=180\cdot100e^{K+4}\,e^{2(K+2)}N^{-1/32}\log N\longrightarrow0.
 \label{eq:unit-parameters}
\end{align}
In \eqref{eq:unit-block-length} we use the mean value theorem for the
eighth power on $[j,j+1]$, $j^7=N^{7/8}$, $j\ge60$ and $8(61/60)^7<9$,
and \eqref{eq:unit-amplitude} follows from it. In the last three bounds
we substitute \eqref{eq:unit-amplitude} and $d_0=N^{3/2-1/64}$.

\emph{Step 2: selected disks.}
We call an annular root $\alpha$, that is, a root of $f_N$ with
$||\alpha|-1|\le K/N$, good if $|f_N'(\alpha)|\ge N^{3/2}/L$.
If $|z-\alpha|\le4a/d_0$, then
$|z|\le|\alpha|+4a/d_0\le1+K/N+1/N=1+(K+1)/N<\rho$, since $4a/d_0\le1/N$
by Step~1.
Hence, outside the exceptional event of
Lemma~\ref{lem:second-derivative}, we have $|f_N''|\le M_2$ throughout
this disk, and Lemma~\ref{lem:isolation-interface}, applied with
$P=f_N$, $|f_N'(\alpha)|\ge d_0$ and $4M_2a\le d_0^2$, shows that the
open disk of radius $2a/d_0$ about $\alpha$ contains exactly one zero,
that $|f_N|\ge3a/2$ on its boundary, and that these disks, for the family
of good annular roots with the common $a$, $d_0$ and $M_2$, are pairwise
disjoint. We call them the selected disks.

\emph{Step 3: unselected roots.}
Let $U_N$ denote the number of unselected roots.
Since every good annular root selects a disk that contains no other
root, the unselected roots are the roots with $||\alpha|-1|>K/N$ and the
annular roots that are not good, which $B_{K,N}$ counts. Consequently,
\begin{align}
 U_N
 &\le\#\{\alpha:f_N(\alpha)=0,\ ||\alpha|-1|>K/N\}+B_{K,N},\nonumber\\
 \frac{U_N}{N}
 &\le\frac{2\log2}{K}+C(K)N^{-1/16}(\log N)^7+C(K)N^{-1/16}\log N\nonumber\\
 &=\frac{2\log2}{K}+O_K\bigl(N^{-1/16}(\log N)^7\bigr),
 \label{eq:unit-unselected}
\end{align}
where the first count is bounded by \eqref{eq:sign-annular-mass} of
Corollary~\ref{cor:radial-probability}, and, by
Lemma~\ref{lem:small-derivatives}, $B_{K,N}$ is at most
$C(K)NL^{-4}\log N$, which is $C(K)N^{15/16}\log N$, outside the
exceptional events of these two results.

\emph{Step 4: the target circles.}
The profile in Proposition~\ref{prop:profiles} satisfies
\[
 F'=\Phi,\qquad 0\le\Phi\le1,\qquad
 0\le\Phi'=2\operatorname{Var}_x(t)\le\frac12.
\]
Indeed, $\Phi(x)$ is the mean of $t\in[0,1]$ under the density
proportional to $e^{2xt}$, and $\Phi'(x)=2\operatorname{Var}_x(t)$ by
\eqref{eq:Phi-variance}, where the variance is at most the mean square
deviation from $\frac12$:
\[
 \operatorname{Var}_x(t)
 \le\frac{\int_0^1(t-\frac12)^2e^{2xt}\dd t}{\int_0^1e^{2xt}\dd t}
 \le\frac14.
\]

We use the logarithmic bounds at the radii $1+(x+qh)/N$, $|q|\le2$,
about $r_N=1+x/N$. The offset $h$ must meet the three requirements of
Table~\ref{tab:moving-offset}, and $h=N^{-1/64}$ does, since the entries
of its last column tend to zero.

We apply Theorem~\ref{thm:sign-logarithms}, with $R$ in place of $K$, at
the five probes $r_q=1+y_q/N$, where $y_q=x+qh$ and $q=-2,\dots,2$.
They lie in $[1-R/N,1+R/N]$, since $|y_q|\le|x|+2h<R$. For all large
$N$, outside five exceptional events, one at each probe, the theorem
gives
$|J_N(r_q)-\log\sigma_N(r_q)+\gamma/2|\le C(R)N^{-1/16}(\log N)^7$, and,
for $N\ge2R$, Lemma~\ref{lem:profile-rate} with $R$ in place of $K$ gives
$|\log\sigma_N(r_q)-\frac12\log N-F(y_q)|\le(R^2+e^{4R}/2)/N$. Adding the
two bounds, and using $1/N\le N^{-1/16}(\log N)^7$ for $N\ge3$, we get,
outside the five exceptional events,
\begin{align}
 \left|J_N(r_q)-\frac12\log N+\frac\gamma2-F(y_q)\right|
 &\le C(R)N^{-1/16}(\log N)^7+\frac{R^2+e^{4R}/2}N\nonumber\\
 &\le\eta_N:=(C(R)+R^2+e^{4R}/2)N^{-1/16}(\log N)^7.
 \label{eq:moving-probes}
\end{align}
For $q=-1,0,1$ we apply Lemma~\ref{lem:local-radial-secant} at $y_q$,
with $R$ in place of $K$, the centering $\frac12\log N$, the profile $F$,
$M_F=1/2$ and $B_F=1$ (by the bounds on $\Phi=F'$ above), and the
logarithmic and profile errors $\eta_{\log}=C(R)N^{-1/16}(\log N)^7$ and
$\varepsilon=(R^2+e^{4R}/2)/N$ of \eqref{eq:moving-probes}, so that
$\eta_{\log}+\varepsilon\le\eta_N$. Its hypotheses hold outside the five
exceptional events, since:
\begin{itemize}
\item $J_N(r_q)$ is finite there, so that $f_N\ne0$;
\item $|y_q|+h\le|x|+2h<R<N$, the last since $N\ge2R$;
\item its three radii $1+(y_q\mp h)/N$ and $r_q$ are among the five
probes.
\end{itemize}
It gives
\[
 \left|\frac{\nu_N(r_q)}N-\Phi(y_q)\right|
 \le\frac h4+\frac RN+\frac{2\eta_N(1+R/N)}h.
\]
Since $r_N+qh/N=r_q$ and $|\Phi(y_q)-\Phi(x)|\le|q|h/2$ by
$0\le\Phi'\le1/2$, and since $|q|h/2+h/4\le2h$ for $|q|\le1$ and
$1+R/N\le2$, we get, for $q=-1,0,1$,
\begin{align}
 \left|\frac{\nu_N(r_N+qh/N)}N-\Phi(x)\right|
 &\le\frac{|q|h}{2}+\frac h4+\frac RN
       +\frac{2\eta_N(1+R/N)}h\nonumber\\
 &\le\varepsilon_N:=2h+\frac RN+\frac{4\eta_N}h\nonumber\\
 &=2N^{-1/64}+\frac RN+4(C(R)+R^2+e^{4R}/2)N^{-3/64}(\log N)^7\nonumber\\
 &=O_R\bigl(N^{-1/64}\bigr).
 \label{eq:moving-secant}
\end{align}
Explicit constants for this estimate are given in \eqref{eq:E1-local}.
Subtracting the two endpoint counts and using \eqref{eq:moving-secant}
with $q=\pm1$, we get
\begin{align}
 &\#\{\alpha:f_N(\alpha)=0,\ ||\alpha|-r_N|<h/N\}\nonumber\\
 &\qquad=\#\{\alpha:f_N(\alpha)=0,\ |\alpha|<r_N+h/N\}-\nu_N(r_N-h/N)\nonumber\\
 &\qquad\le\nu_N(r_N+h/N)-\nu_N(r_N-h/N)\nonumber\\
 &\qquad=N\left(\frac{\nu_N(r_N+h/N)}N-\Phi(x)\right)
   +N\left(\Phi(x)-\frac{\nu_N(r_N-h/N)}N\right)\nonumber\\
 &\qquad\le2\varepsilon_NN,
 \label{eq:moving-band}
\end{align}
where the first equality holds since the inner inequality is strict.

For $N\le n\le N+m_j$ the target radius moves by
$|(1+x/n)-(1+x/N)|=|x|(n-N)/(nN)\le|x|m_j/N^2$, and $m_j\le9N^{7/8}$ for
$j\ge60$ by \eqref{eq:unit-block-length}. Since $N/h=N^{65/64}$, these
bounds and \eqref{eq:unit-radius} yield
\begin{equation}
 \begin{aligned}
 \frac Nh\left(\frac{2a}{d_0}+\frac{|x|m_j}{N^2}\right)
 &\le90e^{2(K+2)}N^{-67/64+65/64}\sqrt{\log N}+9|x|N^{7/8-2+65/64}\\
 &=90e^{2(K+2)}N^{-1/32}\sqrt{\log N}+9|x|N^{-7/64}
 \longrightarrow0.
 \end{aligned}
 \label{eq:unit-radius-ratio}
\end{equation}
Hence $2a/d_0+|x|m_j/N^2<h/N$ for all large $N$, depending on $K$ and
$x$. Let $D$ be a selected disk whose closure meets the closed annulus
between the circles of radii $r_N$ and $1+x/n$, let $\alpha$ be its
center, the root of $f_N$ in it, and let $z$ be a point of this annulus
in the closure of $D$. Then $|z-\alpha|\le2a/d_0$ and $|z|$ lies between
$r_N$ and $1+x/n$, so that
\begin{align*}
 \bigl||\alpha|-r_N\bigr|
 &\le\bigl||\alpha|-|z|\bigr|+\bigl||z|-r_N\bigr|
 \le|\alpha-z|+\bigl||z|-r_N\bigr|\\
 &\le\frac{2a}{d_0}+\left|\left(1+\frac xn\right)-r_N\right|
 \le\frac{2a}{d_0}+\frac{|x|m_j}{N^2}<\frac hN.
\end{align*}
Hence the root in such a disk lies in the strict band of
\eqref{eq:moving-band}. Only such disks can carry matched roots to
different sides of their target circles, since a disk whose closure
misses the closed annulus lies inside both circles or outside both
(proof of Corollary~\ref{cor:T4-two-radii}). Distinct disks have
distinct roots, so by \eqref{eq:moving-band} the number
$c(r_N,1+x/n)$ of these disks is at most $2\varepsilon_NN$.
For $x=0$ the two target circles are the unit circle, the annulus
between them is that circle, and the band is
$1-N^{-65/64}<|\alpha|<1+N^{-65/64}$.

\emph{Step 5: comparison within a block.}
For $N<n\le N+m_j$, we write $f_n=f_N+g_{n-N}$, with $g_m$ as in
Lemma~\ref{lem:tails} at width $K+2$. That lemma bounds all the partial
tails at once, so outside the maximal-tail event $|g_{n-N}|<a$ on
$|z|\le\rho$ for all these $n$. This disk contains the closed selected
disks (Step~2), on whose boundaries $|f_N|\ge3a/2>a$. Hence the selected
disks satisfy the hypotheses of Theorem~\ref{thm:T4} with $P=f_N$,
$G=g_{n-N}$ and $Q=f_n$, for the entire block and all target radii at
once. Moreover, $\deg f_n=n$ and $\deg f_N=N$, since
$\epsilon_n,\epsilon_N\ne0$, and each of the $s$ selected disks contains
exactly one root of $f_N$, the others being unselected, so that
$\deg f_N-s=N-s=U_N$.
By Corollary~\ref{cor:T4-two-radii}, with $P=f_N$, $Q=f_n$, $r=r_N$ and
$r'=1+x/n$, and by Step~4, we have
\begin{align*}
 &\max_{N\le n\le N+m_j}|\nu_n(1+x/n)-\nu_N(r_N)|\\
 &\qquad\le\max_{N\le n\le N+m_j}\bigl\{c(r_N,1+x/n)+(\deg f_N-s)
   +\max(0,\deg f_n-\deg f_N)\bigr\}\\
 &\qquad=\max_{N\le n\le N+m_j}\bigl\{c(r_N,1+x/n)+U_N+(n-N)\bigr\}
 \le2\varepsilon_NN+U_N+m_j,
\end{align*}
which holds trivially for $n=N$.
Splitting $\nu_n(1+x/n)/n-\Phi(x)$ into three terms and using
\eqref{eq:moving-secant} with $q=0$, the matching bound above and
$0\le\Phi\le1$, we get, for $N\le n\le N+m_j$,
\begin{align}
 \left|\frac{\nu_n(1+x/n)}n-\Phi(x)\right|
 &=\biggl|\frac Nn\left(\frac{\nu_N(r_N)}N-\Phi(x)\right)\nonumber\\
 &\qquad+\frac{\nu_n(1+x/n)-\nu_N(r_N)}n-\frac{n-N}n\Phi(x)\biggr|\nonumber\\
 &\le \frac Nn\left|\frac{\nu_N(r_N)}N-\Phi(x)\right|
 +\frac{|\nu_n(1+x/n)-\nu_N(r_N)|}{n}\nonumber\\
 &\qquad+\frac{n-N}{n}\Phi(x)\nonumber\\
 &\le\varepsilon_N+\frac{2\varepsilon_NN+U_N+m_j}N+\frac{m_j}N
 =3\varepsilon_N+\frac{U_N}N+\frac{2m_j}N.
 \label{eq:unit-block}
\end{align}

\emph{Step 6: exceptional events, and $K\to\infty$.}
Let $E_{K,R,x,j}$ denote the common exceptional event in block $j$, the
union of the events of Table~\ref{tab:unit-events}, where $\Pi_K(N)$ is
the right-hand side of \eqref{eq:sign-log-concentration} and $\Pi_R(N)$
is the same with $R$ in place of $K$. For all large $j$,
\begin{equation}
 \Prob(E_{K,R,x,j})
 \le4\Pi_K(N)+5\Pi_R(N)+N^{-3}+2N^{-10}+C(K)N^{-5/16}(\log N)^2.
                                                        \label{eq:unit-failure}
\end{equation}
Here the shared local-count and derivative events are counted once each,
and the maximal-tail event once for the whole block, since outside it
Step~5 holds for all degrees of the block and all target radii.
With $N=j^8$ we have
$\Pi_K(j^8)=C(K)j^{-3}+j^{-96}+C(K)j^{-3}(8\log j)^{-12}$, since
$N^{-3/8}=j^{-3}$, $N^{-12}=j^{-96}$ and $\log N=8\log j$, and
\eqref{eq:unit-failure} reads
\[
 \Prob(E_{K,R,x,j})
 \le4\Pi_K(j^8)+5\Pi_R(j^8)+j^{-24}+2j^{-80}+C(K)j^{-5/2}(8\log j)^2.
\]
For fixed $K$, $R$ and $x$, every term is summable over $j$, since the
exponents of $j$ are $-3$, $-96$, $-24$, $-80$ and $-5/2$, and powers of
$\log j$ do not affect convergence.
In finite form, \eqref{eq:unit-failure} is \eqref{eq:block-failure}, and
the root-count bound of Step~3 is \eqref{eq:block-unselected}.

Let $\Omega_{K,R,x}=(\limsup_jE_{K,R,x,j})^c$.
Since $\limsup_jE_{K,R,x,j}\subset\bigcup_{j\ge J}E_{K,R,x,j}$ for every
$J$, the summability above gives
$\Prob(\Omega_{K,R,x}^c)\le\lim_{J\to\infty}\sum_{j\ge J}\Prob(E_{K,R,x,j})=0$.
On $\Omega_{K,R,x}$ the preceding bounds hold for all large $j$, and
hence, by \eqref{eq:unit-block}, \eqref{eq:unit-unselected} and
$m_j\le9N^{7/8}$, for all large $j$ and $N\le n\le N+m_j$ we get
\[
 \left|\frac{\nu_n(1+x/n)}n-\Phi(x)\right|
 \le3\varepsilon_N+\frac{2\log2}K+C(K)N^{-1/16}(\log N)^7+18N^{-1/8}.
\]
Every large $n$ lies in such a block, and each term on the right
other than $2\log2/K$ tends to zero as $j\to\infty$.
Hence, on the intersection $\Omega_x$ of the events $\Omega_{K,R,x}$ over
integers $K\ge R$, which has probability one as a countable intersection
of events of probability one,
\[
 0\le\limsup_{n\to\infty}
       \left|\frac{\nu_n(1+x/n)}n-\Phi(x)\right|
 \le\frac{2\log2}{K}\longrightarrow0
 \qquad(K\to\infty).
\]

\emph{Step 7: all $x$.}
As a countable intersection, the rational event
$\bigcap_{x\in\mathbb Q}\Omega_x$ also has probability one.

For any real $x$, choose rationals $x_-<x<x_+$. By monotonicity we have
$\nu_n(1+x_-/n)/n\le\nu_n(1+x/n)/n\le\nu_n(1+x_+/n)/n$, and taking
limits on the rational event we get
\[
 \Phi(x_-)\le\liminf_{n\to\infty}\frac{\nu_n(1+x/n)}n
 \le\limsup_{n\to\infty}\frac{\nu_n(1+x/n)}n
 \le\Phi(x_+).
\]
Letting $x_-$ increase to $x$ and $x_+$ decrease to $x$ through
rationals, the continuity of $\Phi$ shows that the limit inferior and the
limit superior are both $\Phi(x)$.
To prove the uniform convergence, fix a compact interval $I$ and
$\delta>0$, write $\widehat\Phi_n(x)=\nu_n(1+x/n)/n$, and choose rational
numbers $x_0<x_1<\cdots<x_M$ with $I\subset(x_0,x_M)$ and
$x_{i+1}-x_i<\delta$. For $x\in[x_i,x_{i+1}]\cap I$ we have
\[
 \widehat\Phi_n(x_i)-\Phi(x_{i+1})
 \le\widehat\Phi_n(x)-\Phi(x)
 \le\widehat\Phi_n(x_{i+1})-\Phi(x_i),
\]
since $\widehat\Phi_n$ is nondecreasing, as $\nu_n$ is, and $\Phi$ is
nondecreasing, as $\Phi'\ge0$.
Taking the supremum over $x\in I$, we get
\begin{align*}
 \sup_{x\in I}|\widehat\Phi_n(x)-\Phi(x)|
 &\le \max_{0\le i\le M}|\widehat\Phi_n(x_i)-\Phi(x_i)|
       +\max_{0\le i<M}\{\Phi(x_{i+1})-\Phi(x_i)\}\\
 &\le \max_{0\le i\le M}|\widehat\Phi_n(x_i)-\Phi(x_i)|
       +\frac{\delta}{2},
\end{align*}
where the second inequality follows from $|\Phi'|\le1/2$ and
$x_{i+1}-x_i<\delta$ by the mean value theorem.
On the rational event the maximum tends to zero, and we then let
$\delta$ tend to zero.
Finally, we intersect over intervals with integer endpoints.
All these intersections take place on the given probability space.
Since the rational event does not depend on $I$, the uniform convergence
holds on it for every compact interval $I$ at once, and in particular for
$I=[-A,A]$ for every $A>0$, which is the assertion of
Theorem~\ref{thm:radial-profile}.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{thm:stronglaw}]
This is the case $x=0$ of Theorem~\ref{thm:radial-profile}, since
$1+0/n=1$ and $\Phi(0)=1/2$.
In the proof above, at $x=0$ we take $R=2$, so that $r_N=1$, both target
circles are the unit circle, and the probes lie at the radii
$1+qN^{-65/64}$, $q=-2,\dots,2$. For every integer $K\ge2$ and all large
$j$, outside $E_{K,2,0,j}$ and uniformly for $N\le n\le N+m_j$,
\eqref{eq:unit-block} reads
\begin{equation}
 \left|\frac{\nu_n(1)}n-\frac12\right|
 \le3\varepsilon_N+\frac{U_N}N+\frac{2m_j}N,
 \label{eq:unit-circle-block}
\end{equation}
where $\varepsilon_N=2N^{-1/64}+2/N+4(C(2)+4+e^{8}/2)N^{-3/64}(\log N)^7$
does not depend on $K$.
\end{proof}

\begin{remark}[A deterministic block envelope]\label{rem:block-envelope}
There are deterministic numbers $\varepsilon_j\to0$ and events $E_j$,
defined on the original probability space, such that
$\sum_j\Prob(E_j)<\infty$ and, off $E_j$,
\[
 \max_{j^8\le n<(j+1)^8}\frac{|\nu_n(1)-\nu_{j^8}(1)|}{j^8}\le\varepsilon_j.
\]
\end{remark}

\begin{proof}
We apply Lemma~\ref{lem:deterministic-diagonalization} with the exponent
$8$, $n_j=j^8$ and $X_n=\nu_n(1)$, for which its conclusion is the
assertion of the remark. It requires, for each positive integer $k$,
numbers $a_k$, $b_k(j)$ and $p_k(j)$ such that, for every $j$,
\[
 \Prob\left\{\max_{n_j\le n<n_{j+1}}\frac{|X_n-X_{n_j}|}{n_j}
 >a_k+b_k(j)\right\}\le p_k(j),
\]
where $a_k\to0$ as $k\to\infty$, and where $b_k(j)\to0$ as $j\to\infty$
and $\sum_jp_k(j)<\infty$ for each fixed $k$.

Let $k$ be a positive integer, and put $K=k+1$, so that $K\ge2=R$, as the
proof of Corollary~\ref{thm:stronglaw} above requires at $x=0$. Since every
threshold in that proof depends only on $K$, there is a deterministic
$j_0(K)\ge60$ such that for every $j\ge j_0(K)$ the bound
\eqref{eq:unit-failure} with $R=2$ and $x=0$ holds and, outside the event
$E_{K,2,0,j}$, with $N=j^8$, the matching bound of Step~5,
\eqref{eq:unit-unselected}, the bound $\varepsilon_N\le C_0N^{-1/64}$ with
an absolute constant $C_0$, and \eqref{eq:unit-block-length} give
\begin{align*}
 \max_{N<n\le N+m_j}|\nu_n(1)-\nu_N(1)|&\le2\varepsilon_NN+U_N+m_j,\\
 \frac{U_N}N&\le\frac{2\log2}K+C(K)N^{-1/16}(\log N)^7,\\
 2\varepsilon_N&\le2C_0N^{-1/64},\\
 \frac{m_j}N&\le9N^{-1/8}.
\end{align*}
We set $a_k=2\log2/K$,
\[
 b_k(j)=C(K)N^{-1/16}(\log N)^7+2C_0N^{-1/64}+9N^{-1/8},
\]
and let $p_k(j)$ be the bound \eqref{eq:unit-failure} for
$\Prob(E_{K,2,0,j})$ when $j\ge j_0(K)$, and $p_k(j)=1$ when
$j<j_0(K)$.
Since the block $n_j\le n<n_{j+1}$ is contained in $N\le n\le N+m_j$,
and the increment vanishes at $n=N$, the four bounds above give, for
$j\ge j_0(K)$ and outside $E_{K,2,0,j}$,
\begin{align*}
 \max_{n_j\le n<n_{j+1}}\frac{|X_n-X_{n_j}|}{n_j}
 &\le\frac1N\max_{N<n\le N+m_j}|\nu_n(1)-\nu_N(1)|\\
 &\le2\varepsilon_N+\frac{U_N}N+\frac{m_j}N
 \le a_k+b_k(j).
\end{align*}
Hence the probability in the above requirement is at most
$\Prob(E_{K,2,0,j})\le p_k(j)$ when $j\ge j_0(K)$, and at most $1=p_k(j)$
when $j<j_0(K)$. Moreover, $a_k\to0$ as $k\to\infty$, and, for fixed
$k$, $b_k(j)\to0$ as $j\to\infty$ and $\sum_jp_k(j)<\infty$, since only
finitely many of its terms equal one and the others are summable by
Step~6 of the proof of Theorem~\ref{thm:radial-profile}.
By Lemma~\ref{lem:deterministic-diagonalization}, there are therefore
deterministic $\varepsilon_j\to0$ and events $E_j$ with
$\sum_j\Prob(E_j)<\infty$, off which the normalized block increment is at
most $\varepsilon_j$. We may take $E_j$ to be the event on which the
normalized block increment exceeds $\varepsilon_j$, since it is contained
in the event given by the lemma. (In the proof of
Lemma~\ref{lem:deterministic-diagonalization} the given events are this
event for $j\ge J_1$ and $\Omega$ for $j<J_1$, where $J_1$ is the first of
the integers $J_k$ chosen there, and $\sum_{j\ge J_1}\Prob(E_j)\le1$.)
All the events here are defined through the random signs on
the original probability space.
\end{proof}

With $n_j=j^8$ and $X_n=\nu_n(1)$, the numbers $\varepsilon_j$ and the
events $E_j$ of Remark~\ref{rem:block-envelope} are those required by
Theorem~\ref{thm:T5}.
